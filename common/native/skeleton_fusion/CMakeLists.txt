cmake_minimum_required(VERSION 3.20)
project(herta_skeleton_fusion LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The skeleton fusion ABI targets Windows x64 MSVC")
endif()
add_library(herta_skeleton_fusion_windows_x64 SHARED neighbors.cpp)
target_compile_features(herta_skeleton_fusion_windows_x64 PRIVATE cxx_std_17)
target_compile_options(herta_skeleton_fusion_windows_x64 PRIVATE /O2 /fp:strict)
set_property(TARGET herta_skeleton_fusion_windows_x64 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
